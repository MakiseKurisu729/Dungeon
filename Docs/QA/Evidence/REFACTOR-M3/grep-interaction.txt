【复现命令（仓库根目录，PowerShell）】
  $pat = 'Input\.|SelectionManager|UIManager'
  $files = 'Assets\_Project\Scripts\System\Chest.cs','Assets\_Project\Scripts\Units\DownedRescue.cs','Assets\_Project\Scripts\System\DungeonDoor.cs'
  foreach ($f in $files) {
    Get-Content $f | Select-String -Pattern $pat | ForEach-Object {
      $trim = $_.Line.TrimStart()
      $kind = if ($trim.StartsWith('//')) { 'COMMENT(排除)' } else { 'CODE(计入)' }
      "$f : $($_.LineNumber) : $kind : $trim"
    }
  }
  # 只数代码行：
  foreach ($f in $files) {
    (Get-Content $f | Where-Object { -not $_.TrimStart().StartsWith('//') } | Select-String -Pattern $pat).Count
  }
